name: Auto Update Deals

on:
  schedule:
    # تشغيل تلقائي كل 30 دقيقة في السحابة
    - cron: '*/30 * * * *'
  workflow_dispatch: # يتيح لك التشغيل اليدوي بضغطة زر عند الحاجة

jobs:
  scrape-and-update:
    runs-on: ubuntu-latest
    permissions:
      contents: write

    steps:
      - name: تحميل ملفات المشروع
        uses: actions/checkout@v4

      - name: إعداد بيئة بايثون
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: تثبيت المتطلبات وPlaywright
        run: |
          pip install -r requirements.txt
          playwright install --with-deps chromium

      - name: تشغيل الروبوت لسحب الصفقات
        run: python bot.py

      - name: حفظ وتحديث ملف deals.js في الموقع تلقائياً
        run: |
          git config --global user.name 'DealsBot'
          git config --global user.email 'bot@deals.local'
          git add deals.js
          git diff --quiet && git diff --staged --quiet || (git commit -m "Auto-update deals [skip ci]" && git push)
